% Fig. 1: pipeline of the learned residual merge cost (L).
% Needs: \usepackage{tikz} \usetikzlibrary{positioning,calc,arrows.meta,fit,backgrounds}
% Use inside a two-column figure:
%   \begin{figure*}[t]\centering\input{fig_pipeline}\caption{...}\label{fig:pipeline}\end{figure*}
\definecolor{lblue}{HTML}{2A78D6}
\definecolor{lbluefill}{HTML}{E3EEFB}
\definecolor{lgray}{HTML}{52514E}
\definecolor{lgrayfill}{HTML}{F1F0EC}
\definecolor{lpta}{HTML}{EB6834}
\definecolor{lptafill}{HTML}{FDEDE6}
\begin{tikzpicture}[
  font=\footnotesize,
  >={Stealth[length=2.2mm]},
  box/.style={draw=lgray, fill=lgrayfill, rounded corners=2pt, align=center,
              minimum height=12mm, text width=25mm, inner sep=3pt, line width=0.6pt},
  learned/.style={box, draw=lblue, fill=lbluefill, line width=0.9pt},
  pta/.style={box, draw=lpta, fill=lptafill},
  flow/.style={->, line width=0.7pt, draw=lgray},
  lflow/.style={->, line width=0.9pt, draw=lblue},
  note/.style={font=\scriptsize, text=lgray, align=center},
  node distance=4mm and 5mm
]
% ---------------------------------------------------------------- simulation, one round
\node[box] (atoms) {\textbf{Branches at round $t$}\\ weighted states\\ $(w_i,\rho_i)$};
\node[box, right=of atoms] (prop) {\textbf{Exact propagation}\\ one purification\\ round; one branch per\\ accepted outcome};
\node[box, right=of prop] (cand) {\textbf{Candidate pairs}\\ 8 nearest neighbours\\ in Pauli coordinates};
\node[learned, right=of cand] (cost) {\textbf{Learned cost}\\ pairwise MLP:\\ $c_{ab}$ = error left\\ after PTA correction};
\node[box, right=of cost] (merge) {\textbf{Budgeted merging}\\ cheapest pairs first,\\ until $\sum c_{ab} = \alpha\,\epsilon/40$\\ (cap: 256 branches)};

\draw[flow] (atoms) -- (prop);
\draw[flow] (prop) -- (cand);
\draw[flow] (cand) -- (cost);
\draw[lflow] (cost) -- (merge);
% loop back to the next round
\draw[flow] (merge.north) -- ++(0,5mm) -| node[note, pos=0.25, above] {next round $t+1$ (40 rounds)} (atoms.north);

% readout with the PTA correction
\node[pta, below=7mm of merge] (corr) {\textbf{PTA correction}\\ add the merge bias\\ that PTA predicts};
\node[box, right=5mm of corr, text width=17mm, fill=white] (out) {\textbf{Output}\\ $P_t$, $F_t$};
\draw[flow] (merge) -- (corr);
\draw[flow] (corr) -- (out);

% ---------------------------------------------------------------- offline training
\node[box, anchor=north] (pairs) at ($(atoms.south |- corr.south)+(0,-11mm)$)
  {\textbf{Training pairs}\\ branch pairs at\\ merge times (48\\ training preparations)};
\node[box, right=of pairs] (labels) {\textbf{Exact labels}\\ true merge gap\\ (exact tree + coupled\\ Monte Carlo)};
\node[pta, right=of labels] (resid) {\textbf{Residual target}\\ true gap $-$ PTA gap,\\ for $P$ and $F$ in\\ 4 horizon blocks};
\node[learned, right=of resid] (train) {\textbf{Train MLP}\\ regression + ranking\\ + calibration loss};

\draw[flow] (pairs) -- (labels);
\draw[flow] (labels) -- (resid);
\draw[flow] (resid) -- (train);
\draw[lflow] (train.north) -- node[note, right, text=lblue] {frozen model} (cost.south);

% ---------------------------------------------------------------- frames and labels
\begin{scope}[on background layer]
  \node[draw=lgray!60, dashed, rounded corners=4pt, inner sep=3mm,
        fit=(atoms) (prop) (cand) (cost) (merge) (corr) (out)] (simframe) {};
  \node[draw=lgray!60, dashed, rounded corners=4pt, inner sep=3mm,
        fit=(pairs) (labels) (resid) (train)] (trainframe) {};
\end{scope}
\node[note, anchor=south west, font=\scriptsize\bfseries] at ($(simframe.north west)+(1mm,0.5mm)$) {Simulation (deployment)};
\node[note, anchor=north west, font=\scriptsize\bfseries] at ($(trainframe.south west)+(1mm,-0.5mm)$) {Offline training};

% legend
\node[learned, minimum height=3mm, text width=4mm, inner sep=1pt, right=6mm of train] (lg1) {};
\node[note, right=1mm of lg1, align=left] {learned};
\node[pta, minimum height=3mm, text width=4mm, inner sep=1pt, below=1.5mm of lg1] (lg2) {};
\node[note, right=1mm of lg2, align=left] {uses PTA};
\node[box, minimum height=3mm, text width=4mm, inner sep=1pt, below=1.5mm of lg2] (lg3) {};
\node[note, right=1mm of lg3, align=left] {exact / fixed};
\end{tikzpicture}
